name=OBC-STD Minifs (AIR ESIEA)
version=0.1.2

# ------------------------------------------------------------------ toolchain
compiler.warning_flags=-w
compiler.warning_flags.none=-w
compiler.warning_flags.default=
compiler.warning_flags.more=-Wall
compiler.warning_flags.all=-Wall -Wextra

compiler.path={runtime.tools.arm-none-eabi-gcc-7-2017q4.path}/bin/
compiler.c.cmd=arm-none-eabi-gcc
compiler.cpp.cmd=arm-none-eabi-g++
compiler.S.cmd=arm-none-eabi-gcc
compiler.ar.cmd=arm-none-eabi-ar
compiler.objcopy.cmd=arm-none-eabi-objcopy
compiler.elf2hex.cmd=arm-none-eabi-objcopy
compiler.size.cmd=arm-none-eabi-size

compiler.mcu.flags=-mcpu=cortex-m4 -mthumb -mfloat-abi=hard -mfpu=fpv4-sp-d16

compiler.obc.includes="-I{build.core.path}" "-I{build.core.path}/cmsis/Core/Include" "-I{build.core.path}/cmsis/Device/samd51/include" "-I{build.core.path}/tinyusb/src"

compiler.c.flags={compiler.mcu.flags} -c -g -Os {compiler.warning_flags} -std=gnu11 -ffunction-sections -fdata-sections -nostdlib --param max-inline-insns-single=500 -MMD
compiler.cpp.flags={compiler.mcu.flags} -c -g -Os {compiler.warning_flags} -std=gnu++11 -ffunction-sections -fdata-sections -fno-threadsafe-statics -nostdlib --param max-inline-insns-single=500 -fno-rtti -fno-exceptions -MMD
compiler.S.flags={compiler.mcu.flags} -c -g -x assembler-with-cpp -MMD
compiler.ar.flags=rcs
compiler.c.elf.flags={compiler.mcu.flags} -Os -Wl,--gc-sections -save-temps --specs=nano.specs -u _printf_float
compiler.elf2bin.flags=-O binary
compiler.elf2hex.flags=-O ihex -R .eeprom
compiler.size.flags=-A

# These can be overridden in platform.local.txt
compiler.c.extra_flags=
compiler.cpp.extra_flags=
compiler.S.extra_flags=
compiler.ar.extra_flags=
compiler.c.elf.extra_flags=
compiler.elf2hex.extra_flags=

build.usb_flags=-DUSB_VID={build.vid} -DUSB_PID={build.pid}
build.extra_flags=-D{build.mcu_define} {build.usb_flags}

# ------------------------------------------------------------------ recipes
recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} -DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} {compiler.c.extra_flags} {build.extra_flags} {compiler.obc.includes} {includes} "{source_file}" -o "{object_file}"

recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} -DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} {compiler.cpp.extra_flags} {build.extra_flags} {compiler.obc.includes} {includes} "{source_file}" -o "{object_file}"

recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.S.flags} -DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} {compiler.S.extra_flags} {build.extra_flags} {compiler.obc.includes} {includes} "{source_file}" -o "{object_file}"

recipe.ar.pattern="{compiler.path}{compiler.ar.cmd}" {compiler.ar.flags} {compiler.ar.extra_flags} "{archive_file_path}" "{object_file}"

recipe.c.combine.pattern="{compiler.path}{compiler.c.cmd}" "-L{build.path}" "-L{build.variant.path}/linker_scripts/gcc" {compiler.c.elf.flags} {compiler.c.elf.extra_flags} "-T{build.variant.path}/{build.ldscript}" "-Wl,-Map,{build.path}/{build.project_name}.map" -o "{build.path}/{build.project_name}.elf" -Wl,--start-group {object_files} "{archive_file_path}" -lm -lc -lgcc -Wl,--end-group

recipe.objcopy.bin.pattern="{compiler.path}{compiler.objcopy.cmd}" {compiler.elf2bin.flags} "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.bin"
recipe.objcopy.hex.pattern="{compiler.path}{compiler.elf2hex.cmd}" {compiler.elf2hex.flags} {compiler.elf2hex.extra_flags} "{build.path}/{build.project_name}.elf" "{build.path}/{build.project_name}.hex"

recipe.output.tmp_file={build.project_name}.bin
recipe.output.save_file={build.project_name}.{build.variant}.bin

recipe.size.pattern="{compiler.path}{compiler.size.cmd}" {compiler.size.flags} "{build.path}/{build.project_name}.elf"
recipe.size.regex=\.text\s+([0-9]+).*
recipe.size.regex.data=^(?:\.data|\.bss)\s+([0-9]+).*

# ------------------------------------------------------------------ upload (USB, UF2 bootloader, SAM-BA protocol)
tools.bossac.path={runtime.tools.bossac-1.8.0-48-gb176eee.path}
tools.bossac.cmd=bossac
tools.bossac.cmd.windows=bossac.exe
tools.bossac.upload.params.verbose=-i -d
tools.bossac.upload.params.quiet=
tools.bossac.upload.params.verify=-v
tools.bossac.upload.params.noverify=
tools.bossac.upload.pattern="{path}/{cmd}" {upload.verbose} --port={serial.port.file} -U -i --offset={upload.offset} -w {upload.verify} "{build.path}/{build.project_name}.bin" -R

# ------------------------------------------------------------------ burn bootloader / upload with a SWD probe
tools.openocd.path={runtime.tools.openocd-0.11.0-arduino2.path}
tools.openocd.cmd=bin/openocd
tools.openocd.cmd.windows=bin/openocd.exe
# The probe is selected in Tools > Programmer (see programmers.txt):
#   {openocd.interface} = OpenOCD interface script, {openocd.transport} = swd / hla_swd
tools.openocd.program.params.verbose=-d2
tools.openocd.program.params.quiet=-d0
tools.openocd.program.pattern="{path}/{cmd}" {program.verbose} -s "{path}/share/openocd/scripts/" -f {openocd.interface} -c "transport select {openocd.transport}" -f target/atsame5x.cfg -c "telnet_port disabled; init; reset halt; program {{build.path}/{build.project_name}.bin} verify reset {upload.offset}; shutdown"
tools.openocd.bootloader.params.verbose=-d2
tools.openocd.bootloader.params.quiet=-d0
tools.openocd.bootloader.pattern="{path}/{cmd}" {bootloader.verbose} -s "{path}/share/openocd/scripts/" -f {openocd.interface} -c "transport select {openocd.transport}" -f target/atsame5x.cfg -c "telnet_port disabled; init; reset halt; atsame5 bootloader 0; program {{runtime.platform.path}/bootloaders/{bootloader.file}} verify reset; shutdown"
